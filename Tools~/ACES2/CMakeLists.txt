cmake_minimum_required(VERSION 3.16)
project(Aces2ReferenceCheck LANGUAGES CXX)
set(ACES_CTL_SOURCE "" CACHE PATH "Checkout of the official Academy CTL interpreter")
if(NOT EXISTS "${ACES_CTL_SOURCE}/lib/IlmCtl/CtlInterpreter.h")
    message(FATAL_ERROR "Set ACES_CTL_SOURCE to the official CTL source checkout")
endif()
set(CTL_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(CTL_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${ACES_CTL_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/ctl")
find_package(OpenEXR 3 CONFIG REQUIRED)
find_package(Imath CONFIG REQUIRED)
add_executable(official-ctl-check official-ctl-check.cpp)
target_compile_features(official-ctl-check PRIVATE cxx_std_17)
target_include_directories(official-ctl-check PRIVATE "${ACES_CTL_SOURCE}/lib/IlmCtl" "${ACES_CTL_SOURCE}/lib/IlmCtlSimd")
target_link_libraries(official-ctl-check PRIVATE IlmCtlSimd IlmCtlMath IlmCtl Imath::Imath OpenEXR::Iex)
